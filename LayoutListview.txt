package com.example.listview

import android.graphics.Color
import androidx.appcompat.app.AppCompatActivity
import android.os.Bundle
import android.widget.ArrayAdapter
import android.widget.Toast
import kotlinx.android.synthetic.main.activity_main.*

class MainActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        var data = arrayOf("RED","GREEN","YELLOW","BLUE","PINK")
        var adp=ArrayAdapter(this,android.R.layout.simple_list_item_1,data)
        listview.adapter=adp
        listview.setOnItemClickListener { adapterView, view, i, l ->
            var d=adapterView.getItemAtPosition(i).toString()
            my_layout.setBackgroundColor(Color.parseColor(d))
        }
    }
}
